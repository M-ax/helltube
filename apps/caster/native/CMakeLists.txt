cmake_minimum_required(VERSION 3.24)
project(helltube_capture LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This helper uses Windows WASAPI. Linux uses PulseAudio/PipeWire utilities.")
endif()
add_executable(helltube-audio windows-audio.cpp)
target_compile_features(helltube-audio PRIVATE cxx_std_17)
target_compile_definitions(helltube-audio PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(helltube-audio PRIVATE ole32 mmdevapi)
set_property(TARGET helltube-audio PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_target_properties(helltube-audio PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/bin")
add_executable(helltube-video windows-video.cpp)
target_compile_features(helltube-video PRIVATE cxx_std_20)
target_compile_definitions(helltube-video PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(helltube-video PRIVATE /utf-8)
target_link_libraries(helltube-video PRIVATE d3d11 dxgi d3dcompiler dwmapi windowsapp gdi32 user32)
set_property(TARGET helltube-video PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_target_properties(helltube-video PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/bin")
include(CTest)
if(BUILD_TESTING)
  add_executable(helltube-video-test windows-video-test.cpp)
  target_compile_features(helltube-video-test PRIVATE cxx_std_20)
  target_compile_definitions(helltube-video-test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(helltube-video-test PRIVATE /utf-8)
  target_link_libraries(helltube-video-test PRIVATE d3d11 dxgi d3dcompiler dwmapi windowsapp gdi32 user32)
  set_property(TARGET helltube-video-test PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  add_test(NAME native-video-gpu COMMAND helltube-video-test)
endif()
